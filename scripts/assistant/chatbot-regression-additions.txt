    [Theory]
    [InlineData("TH")]
    [InlineData("Vinamilk")]
    [InlineData("có đường")]
    public void PartialCatalogCanRefineBeyondTheFirstPage(string fragment)
    {
        var memory = new ScopedCatalogMemory(new("1", null, null), DateTimeOffset.MaxValue,
            [new("1", "1", "Sữa đặc Ông Thọ")], "sua", HasMoreMatches: true);
        Assert.True(AssistantCatalogClarification.TryQuery(memory, fragment, out var query));
        Assert.StartsWith("sua ", query);
        Assert.False(AssistantCatalogClarification.TryQuery(memory with { HasMoreMatches = false }, fragment, out _));
    }
